rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    function isSignedIn() {
      return request.auth != null;
    }

    function isOwner(userId) {
      return isSignedIn() && request.auth.uid == userId;
    }

    function userData() {
      return get(/databases/$(database)/documents/users/$(request.auth.uid)).data;
    }

    function isAdmin() {
      return isSignedIn() && (
        userData().role == 'admin' || userData().isAdmin == true
      );
    }

    function protectedUserFields() {
      return [
        'role','isAdmin','banned','isPro','isMotorista','isEmpresa',
        'proPlan','proSince','proExpiresAt','motoristaExpiresAt',
        'proCancelled','proCancelledAt','proSubscriptionId','proSubscriptionStatus',
        'proPaymentId','proDaysLeft','proExpired','proExpiredAt',
        'proAwardedBy','proAwardMonth','proPrizeDays',
        'referralProcessed','referredBy','referralCount',
        'alloPoints','apLastLogin','apStreak','apStreakLastDate',
        'proBillingProvider','googlePlayProductId','googlePlayPurchaseTokenHash',
        'googlePlayLatestOrderId','googlePlaySubscriptionState',
        'googlePlayAcknowledgementState','googlePlayAutoRenewing',
        'googlePlayTestPurchase','googlePlayExpiresAt','googlePlayEntitled',
        'googlePlayLastVerifiedAt','legacyProExpiresAt','proManualExpiresAt',
        'proPrizeExpiresAt','proReferralExpiresAt'
      ];
    }

    function legacyFinancialFields() {
      return [
        'transactions','banks','cards','categories','goals','debts','benefits',
        'calculatorSimulations','importHistory','cofres','uberJornadas','uberCorridas',
        'uberGastos','uberAbastec','uberVeiculos','cardTransactions'
      ];
    }

    function migrationAffectedFields() {
      return [
        'transactions','banks','cards','categories','goals','debts','benefits',
        'calculatorSimulations','importHistory','cofres','uberJornadas','uberCorridas',
        'uberGastos','uberAbastec','uberVeiculos','cardTransactions',
        'dataSchemaVersion','dataMigratedAt','updatedAt'
      ];
    }

    function ownerUserUpdateIsSafe() {
      let affected = request.resource.data.diff(resource.data).affectedKeys();
      let normalUpdate = !affected.hasAny(legacyFinancialFields());
      let completedMigration = affected.hasOnly(migrationAffectedFields())
        && request.resource.data.dataSchemaVersion == 39
        && !request.resource.data.keys().hasAny(legacyFinancialFields());
      return !affected.hasAny(protectedUserFields())
        && (normalUpdate || completedMigration);
    }

    match /users/{userId} {
      allow read: if isOwner(userId) || isAdmin();
      allow create: if isOwner(userId)
        && !request.resource.data.keys().hasAny(protectedUserFields())
        && !request.resource.data.keys().hasAny(legacyFinancialFields());
      allow update: if isAdmin() || (isOwner(userId) && ownerUserUpdateIsSafe());
      allow delete: if isAdmin();

      match /ap_history/{documentId} {
        allow read: if isOwner(userId) || isAdmin();
        allow create, update, delete: if false;
      }

      match /notification_dispatches/{documentId} {
        allow read, write: if false;
      }

      match /transactions/{documentId} { allow read, write: if isOwner(userId) || isAdmin(); }
      match /banks/{documentId} { allow read, write: if isOwner(userId) || isAdmin(); }
      match /cards/{documentId} { allow read, write: if isOwner(userId) || isAdmin(); }
      match /categories/{documentId} { allow read, write: if isOwner(userId) || isAdmin(); }
      match /goals/{documentId} { allow read, write: if isOwner(userId) || isAdmin(); }
      match /debts/{documentId} { allow read, write: if isOwner(userId) || isAdmin(); }
      match /benefits/{documentId} { allow read, write: if isOwner(userId) || isAdmin(); }
      match /calculatorSimulations/{documentId} { allow read, write: if isOwner(userId) || isAdmin(); }
      match /importHistory/{documentId} { allow read, write: if isOwner(userId) || isAdmin(); }
      match /cofres/{documentId} { allow read, write: if isOwner(userId) || isAdmin(); }
      match /uberJornadas/{documentId} { allow read, write: if isOwner(userId) || isAdmin(); }
      match /uberCorridas/{documentId} { allow read, write: if isOwner(userId) || isAdmin(); }
      match /uberGastos/{documentId} { allow read, write: if isOwner(userId) || isAdmin(); }
      match /uberAbastec/{documentId} { allow read, write: if isOwner(userId) || isAdmin(); }
      match /uberVeiculos/{documentId} { allow read, write: if isOwner(userId) || isAdmin(); }
      match /cardTransactions/{documentId} { allow read, write: if isOwner(userId) || isAdmin(); }
    }

    match /settings/{documentId} {
      allow read: if true;
      allow create, update, delete: if isAdmin();
    }

    match /support/{userId} {
      allow read, write: if isOwner(userId) || isAdmin();

      match /messages/{messageId} {
        allow read: if isOwner(userId) || isAdmin();
        allow create: if isAdmin() || (
          isOwner(userId) && request.resource.data.from == 'user'
        );
        allow update, delete: if isAdmin();
      }

      match /tickets/{ticketId} {
        allow read: if isOwner(userId) || isAdmin();
        allow create: if isAdmin() || (
          isOwner(userId)
          && request.resource.data.userId == userId
          && request.resource.data.status == 'waiting_support'
        );
        allow update: if isAdmin() || (
          isOwner(userId)
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status','updatedAt'])
          && request.resource.data.status == 'waiting_support'
        );
        allow delete: if isAdmin();
      }
    }

    match /admin_messages/{messageId} {
      allow read, create, update, delete: if isAdmin();
    }

    match /ap_ranking/{month} {
      allow read: if isSignedIn();
      allow create, update, delete: if false;
      match /users/{userId} {
        allow read: if isSignedIn();
        allow create, update, delete: if false;
      }
    }

    match /referrals/{code} {
      allow read: if isSignedIn();
      allow create: if isSignedIn()
        && !exists(/databases/$(database)/documents/referrals/$(code))
        && request.resource.data.userId == request.auth.uid;
      allow update, delete: if isAdmin();
    }

    match /coupons/{code} {
      allow read: if isSignedIn();
      allow create, update, delete: if isAdmin();
    }

    match /admin_logs/{logId} {
      allow read, create: if isAdmin();
      allow update, delete: if false;
    }

    // Coleções internas do backend. O Firebase Admin SDK ignora estas regras.
    match /google_play_purchases/{documentId} { allow read, write: if false; }
    match /google_play_rtdn_events/{documentId} { allow read, write: if false; }
    match /notification_devices/{documentId} { allow read, write: if false; }
    match /allofy_conversations/{document=**} { allow read, write: if false; }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
